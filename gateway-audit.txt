import { randomUUID } from 'node:crypto';
import { connection, NextRequest, NextResponse } from 'next/server';
import {
  ACCESS_COOKIE,
  authenticateGatewayRequest,
  gatewayOrigin,
  grantGatewayAccess,
  requireSameOrigin,
} from '@/lib/gateway/server/access';
import { formatErrorResponse, readGatewayJson } from '@/lib/gateway/server/ingress';
import { createGatewayError } from '@/lib/gateway/contracts/errors';

function accessResponse(member: ReturnType<typeof authenticateGatewayRequest>) {
  return NextResponse.json(
    member
      ? { authenticated: true, workspaceId: member.workspaceId, actorId: member.actorId, roles: [member.role] }
      : { authenticated: false },
    { headers: { 'cache-control': 'no-store' } },
  );
}

export async function GET(request: NextRequest) {
  await connection();
  try {
    return accessResponse(authenticateGatewayRequest(request));
  } catch (error) {
    return formatErrorResponse(error, randomUUID()).response;
  }
}

export async function POST(request: NextRequest) {
  try {
    requireSameOrigin(request);
    const body = await readGatewayJson(request, 4096);
    const token = (body as { token?: unknown } | null)?.token;
    if (typeof token !== 'string') {
      throw createGatewayError({ code: 'invalid_request', message: 'Enter a gateway access token.' });
    }
    const session = grantGatewayAccess(token.trim());
    const response = accessResponse(session.member);
    response.cookies.set(ACCESS_COOKIE, session.cookie, {
      httpOnly: true,
      sameSite: 'strict',
      secure: gatewayOrigin(request).protocol === 'https:',
      path: '/',
      maxAge: session.maxAge,
    });
    return response;
  } catch (error) {
    return formatErrorResponse(error, randomUUID()).response;
  }
}

export async function DELETE(request: NextRequest) {
  try {
    requireSameOrigin(request);
    const response = accessResponse(undefined);
    response.cookies.set(ACCESS_COOKIE, '', {
      httpOnly: true,
      sameSite: 'strict',
      secure: gatewayOrigin(request).protocol === 'https:',
      path: '/',
      maxAge: 0,
    });
    return response;
  } catch (error) {
    return formatErrorResponse(error, randomUUID()).response;
  }
}

import { randomUUID } from 'node:crypto';
import type { GatewayEvent } from '@/lib/gateway/contracts/event';
import { GatewayError } from '@/lib/gateway/contracts/errors';
import { withGatewayIngress } from '@/lib/gateway/server/ingress';
import { getGatewayRegistry } from '@/lib/gateway/server/registry';


const eventHeaders = {
  'Content-Type': 'text/event-stream',
  'Cache-Control': 'no-store, no-transform',
  'X-Accel-Buffering': 'no',
};
const HEARTBEAT = Symbol('heartbeat');
const encoder = new TextEncoder();

function eventFrame(event: GatewayEvent): string {
  const id = event.type === 'resync.requested' ? '' : `id: ${event.sequence}\n`;
  return `${id}data: ${JSON.stringify(event)}\n\n`;
}

export const GET = withGatewayIngress(async (req, ctx) => {
  const query = new URL(req.url).searchParams;
  const sessionId = query.get('sessionId') ?? query.get('session_id');
  if (!sessionId || sessionId.length > 256 || sessionId === '*') {
    throw new GatewayError({ code: 'invalid_request', message: 'A session ID is required for events.', requestId: ctx.requestId });
  }

  const registry = getGatewayRegistry();
  const session = await registry.runService.getSession(sessionId, ctx);
  if (!session) {
    throw new GatewayError({ code: 'not_found', message: 'Session not found.', requestId: ctx.requestId });
  }

  const rawCursor = req.headers.get('last-event-id') || query.get('cursor');
  if (rawCursor === null) {
    return new Response(eventFrame({
      id: randomUUID(),
      type: 'resync.requested',
      sessionId,
      sequence: session.lastEventSequence,
      occurredAt: new Date().toISOString(),
      data: { reason: 'A session snapshot is required.', expectedCursor: 0, latestWatermark: session.lastEventSequence },
    }), { headers: eventHeaders });
  }
  const cursor = Number(rawCursor);
  if (!/^\d+$/.test(rawCursor) || !Number.isSafeInteger(cursor)) {
    throw new GatewayError({ code: 'invalid_request', message: 'Event cursor must be a non-negative integer.', requestId: ctx.requestId });
  }

  const disconnect = new AbortController();
  const signal = AbortSignal.any([req.signal, disconnect.signal]);
  const subscription = registry.eventFeed.subscribe(sessionId, cursor, ctx, signal)[Symbol.asyncIterator]();
  let next: Promise<IteratorResult<GatewayEvent>> | undefined;

  const body = new ReadableStream<Uint8Array>({
    start(controller) {
      controller.enqueue(encoder.encode(': connected\n\n'));
    },
    async pull(controller) {
      let timer: ReturnType<typeof setTimeout> | undefined;
      try {
        // Retain the in-flight iterator read when a heartbeat wins. A slow client
        // only pulls one event at a time, so the response does not queue the journal.
        next ??= subscription.next();
        const heartbeat = new Promise<typeof HEARTBEAT>((resolve) => {
          timer = setTimeout(() => resolve(HEARTBEAT), 15_000);
        });
        const item = await Promise.race([next, heartbeat]);
        if (signal.aborted) {
          if (!disconnect.signal.aborted) controller.close();
          return;
        }
        if (item === HEARTBEAT) {
          controller.enqueue(encoder.encode(': heartbeat\n\n'));
          return;
        }
        next = undefined;
        if (!item.done) controller.enqueue(encoder.encode(eventFrame(item.value)));
        if (item.done || item.value.type === 'resync.requested') {
          controller.close();
          disconnect.abort();
          await subscription.return?.();
        }
      } catch {
        if (!disconnect.signal.aborted) controller.close();
        disconnect.abort();
      } finally {
        clearTimeout(timer);
      }
    },
    async cancel() {
      disconnect.abort();
      await subscription.return?.();
    },
  });

  return new Response(body, { headers: eventHeaders });
});

async function listModels(ctx: RequestContext, connectionId: string | undefined) {
  const available = await getGatewayRegistry().modelCatalog.listModels({ connectionId }, ctx);
  const data = available.map(({ id, name, provider, connectionId, capabilities }) => ({
    id, name, provider, connectionId, capabilities,
  }));
  return NextResponse.json({ models: [...new Set(data.map((model) => model.id))], data, total: data.length });
}

function validateFilter(raw: unknown): string | undefined {
  if (raw === undefined || raw === null) return;
  if (typeof raw !== 'string' || !/^[a-zA-Z0-9_-]{1,128}$/.test(raw)) {
    throw createGatewayError({ code: 'invalid_request', field: 'connectionId', message: 'Invalid connection selection.' });
  }
  return raw;
}

function rejectProviderOverrides() {
  throw createGatewayError({ code: 'invalid_request', message: 'Use a saved provider connection.' });
}

export const GET = withGatewayIngress(async (req, ctx) => {
  const query = req.nextUrl.searchParams;
  if (query.has('baseUrl') || query.has('apiKey')) rejectProviderOverrides();
  return listModels(ctx, validateFilter(query.get('connectionId')));
});

export const POST = withGatewayIngress(async (req, ctx) => {
  const body = await readGatewayJson(req, 4096);
  if (body !== null && (typeof body !== 'object' || Array.isArray(body))) {
    throw createGatewayError({ code: 'invalid_request', message: 'Expected a model query object.' });
  }
  const input = body as { apiKey?: unknown; baseUrl?: unknown; connectionId?: unknown } | null;
  if (input && ('baseUrl' in input || 'apiKey' in input)) rejectProviderOverrides();
  return listModels(ctx, validateFilter(input?.connectionId));

export const GATEWAY_ACCESS_CHANGED = "kantha:access-changed";

export class GatewayRequestError extends Error {
  constructor(message: string, public readonly status: number, public readonly code?: string) {
    super(message);
  }
}

export async function readGatewayError(response: Response): Promise<GatewayRequestError> {
  const data = await response.json().catch(() => null);
  const message =
    (typeof data?.message === "string" && data.message) ||
    (typeof data?.error === "string" && data.error) ||
    `Gateway request failed (HTTP ${response.status}).`;
  return new GatewayRequestError(message, response.status, data?.code);
}

export async function gatewayRequest<T>(path: string, options: RequestInit = {}): Promise<T> {
  const response = await fetch(path, { cache: "no-store", ...options });
  if (!response.ok) {
    if (response.status === 401 && path !== "/api/gateway/access") {
      window.dispatchEvent(new Event(GATEWAY_ACCESS_CHANGED));
    }
    throw await readGatewayError(response);
  }
  return response.status === 204 ? (undefined as T) : response.json();
}

export function jsonBody(body: unknown): Pick<RequestInit, "headers" | "body"> {
  return {
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(body),
  };
}

export function gatewayMessage(error: unknown, fallback: string): string {
  return error instanceof GatewayRequestError ? error.message : fallback;
}

"use client";

import { useCallback, useEffect, useRef, useState } from "react";
import type { GatewayEvent, RunStateChangedPayload } from "@/lib/gateway/contracts/event";
import type { GatewayApproval } from "@/lib/gateway/contracts/approval";
import type { GatewayRun, RunSnapshot, RunStatus } from "@/lib/gateway/contracts/run";
import { gatewayMessage, gatewayRequest, GatewayRequestError, jsonBody } from "./gateway-client";

function isTerminal(status: RunStatus) {
  return status === "completed" || status === "cancelled" || status === "failed" || status === "interrupted";
}

export function useGatewayEvents(sessionId: string | undefined, onSnapshot: (snapshot: RunSnapshot) => void) {
  const [run, setRun] = useState<GatewayRun | null>(null);
  const [approval, setApproval] = useState<GatewayApproval | null>(null);
  const [connecting, setConnecting] = useState(false);
  const [error, setError] = useState<string>();
  const [loadedSession, setLoadedSession] = useState<string>();
  const snapshotCallback = useRef(onSnapshot);
  const reload = useRef<() => Promise<RunSnapshot | undefined>>(async () => undefined);

  useEffect(() => {
    snapshotCallback.current = onSnapshot;
  }, [onSnapshot]);

  useEffect(() => {
    if (!sessionId) return;
    const controller = new AbortController();
    let source: EventSource | undefined;
    let retry: ReturnType<typeof setTimeout> | undefined;
    let refreshTimer: ReturnType<typeof setTimeout> | undefined;
    let pending: Promise<RunSnapshot | undefined> | undefined;
    let cursor = 0;
    const path = "/api/gateway/sessions/" + encodeURIComponent(sessionId);

    const refresh = () => {
      if (controller.signal.aborted) return Promise.resolve(undefined);
      if (pending) return pending;
      pending = gatewayRequest<RunSnapshot>(path, { signal: controller.signal }).then((snapshot) => {
        if (controller.signal.aborted) return undefined;
        setRun(snapshot.run);
        setLoadedSession(sessionId);
        setError(undefined);
        setApproval((previous) => {
          if (snapshot.pendingApproval) {
            if (previous?.id === snapshot.pendingApproval.id &&
                previous.status !== "waiting" && snapshot.pendingApproval.status === "waiting") return previous;
            return snapshot.pendingApproval;
          }
          return previous?.sessionId === sessionId && previous.runId === snapshot.run?.id ? previous : null;
        });
        snapshotCallback.current(snapshot);
        return snapshot;
      }).catch((error: unknown) => {
        if (controller.signal.aborted) return undefined;
        setLoadedSession(sessionId);
        if (error instanceof GatewayRequestError && error.status === 404) {
          setRun(null);
          setApproval(null);
          setError("This conversation is unavailable. Start a new conversation to continue.");
        } else {
          setError(gatewayMessage(error, "Could not refresh this conversation."));
        }
        throw error;
      }).finally(() => {
        pending = undefined;
      });
      return pending;
    };
    reload.current = refresh;

    const scheduleSnapshot = () => {
      if (refreshTimer) return;
      refreshTimer = setTimeout(() => {
        refreshTimer = undefined;
        const earlier = pending;
        void (async () => {
          await earlier?.catch(() => undefined);
          await refresh().catch(() => undefined);
        })();
      }, 350);
    };

    const reconnect = () => {
      source?.close();
      if (retry) return;
      setConnecting(true);
      retry = setTimeout(() => {
        retry = undefined;
        void connect();
      }, 1500);
    };

    const connect = async () => {
      try {
        const snapshot = await refresh();
        if (!snapshot || controller.signal.aborted) return;
        cursor = snapshot.watermark;
      } catch (error) {
        if (!controller.signal.aborted && (!(error instanceof GatewayRequestError) || ![401, 403, 404].includes(error.status))) {
          reconnect();
        }
        return;
      }

      const query = new URLSearchParams({ sessionId, cursor: String(cursor) });
      source = new EventSource("/api/gateway/events?" + query.toString());
      source.onopen = () => setConnecting(false);
      source.onerror = reconnect;
      source.onmessage = (message) => {
        let event: GatewayEvent;
        try {
          event = JSON.parse(message.data);
        } catch {
          return;
        }
        if (event.type === "resync.requested") {
          reconnect();
          return;
        }
        if (event.sessionId !== sessionId || event.sequence <= cursor) return;
        cursor = event.sequence;
        if (event.type === "run.state_changed") {
          const { newState } = event.data as RunStateChangedPayload;
          if (newState && event.runId) {
            const runId = event.runId;
            setRun((current) => current?.id === runId ? { ...current, status: newState } : current);
          }
          scheduleSnapshot();
        }
        if (event.type === "approval.requested") {
          setApproval(event.data as GatewayApproval);
          scheduleSnapshot();
        }
        if (event.type === "run.delta" || event.type === "approval.resolved" ||
            event.type === "run.completed" || event.type === "run.failed" || event.type === "run.cancelled") {
          scheduleSnapshot();
        }
      };
    };

    void connect();
    return () => {
      controller.abort();
      source?.close();
      if (retry) clearTimeout(retry);
      if (refreshTimer) clearTimeout(refreshTimer);
      reload.current = async () => undefined;
    };
  }, [sessionId]);

  const approvalId = approval?.id;
  const approvalSession = approval?.sessionId;
  const approvalStatus = approval?.status;
  const approvalExpiry = approval?.expiresAt;
  useEffect(() => {
    if (!approvalId || !approvalExpiry || approvalStatus !== "waiting" || approvalSession !== sessionId) return;
    const controller = new AbortController();
    const expiry = Math.max(Date.parse(approvalExpiry) - Date.now() + 200, 0);
    const timer = setTimeout(() => {
      const path = "/api/gateway/approvals/" + encodeURIComponent(approvalId);
      gatewayRequest<GatewayApproval>(path, { signal: controller.signal })
        .then((latest) => { if (!controller.signal.aborted) setApproval(latest); })
        .catch(() => undefined);
    }, expiry);
    return () => {
      clearTimeout(timer);
      controller.abort();
    };
  }, [approvalId, approvalExpiry, approvalSession, approvalStatus, sessionId]);

  const resolveApproval = useCallback(async (approvalId: string, decision: "approve" | "reject", actionVersion: number) => {
    const path = "/api/gateway/approvals/" + encodeURIComponent(approvalId);
    try {
      const decisionResult = await gatewayRequest<GatewayApproval>(path, {
        method: "POST",
        ...jsonBody({ decision, actionVersion }),
      });
      setApproval(decisionResult);
      void reload.current().catch(() => undefined);
    } catch (error) {
      const observed = await gatewayRequest<GatewayApproval>(path).catch(() => undefined);
      if (observed) setApproval(observed);
      void reload.current().catch(() => undefined);
      if (!observed || observed.status !== (decision === "approve" ? "approved" : "rejected")) {
        throw error;
      }
    }
  }, []);

  const cancel = useCallback(async (runId: string) => {
    const cancelled = await gatewayRequest<GatewayRun>("/api/gateway/runs/" + encodeURIComponent(runId) + "/cancel", {
      method: "POST",
      ...jsonBody({ reason: "Cancelled from Chat" }),
    });
    setRun(cancelled);
    return cancelled;
  }, []);
  const refresh = useCallback(() => reload.current(), []);

  return {
    run: run?.sessionId === sessionId ? run : null,
    approval: approval?.sessionId === sessionId ? approval : null,
    loading: Boolean(sessionId && loadedSession !== sessionId),
    reconnecting: Boolean(sessionId && connecting && (!run || !isTerminal(run.status))),
    error: sessionId ? error : undefined,
    refresh,
    rememberApproval: setApproval,
    resolveApproval,
    cancel,
  };
}

export type GatewayErrorCode =
  | 'invalid_request'
  | 'unauthorized'
  | 'forbidden'
  | 'not_found'

  | 'rate_limited'
  | 'service_unavailable'

  | 'upstream_failed'
  | 'upstream_timeout'
  | 'budget_exceeded'
  | 'subscriber_overflow'
  | 'internal_error';

export interface GatewayErrorEnvelope {
  code: GatewayErrorCode;
  message: string;
  requestId: string;
  retryAfterSeconds?: number;
  field?: string;
  resource?: string;
  details?: unknown;
}

export const GATEWAY_HTTP_STATUS_MAP: Record<GatewayErrorCode, number> = {
  invalid_request: 400,
  unauthorized: 401,
  forbidden: 403,
  not_found: 404,
  conflict: 409,
  rate_limited: 429,
  service_unavailable: 503,
  service_starting: 503,
  service_stopping: 503,
  upstream_failed: 502,
  upstream_timeout: 504,
  budget_exceeded: 402,
  subscriber_overflow: 429,
  internal_error: 500,
};

const gatewayErrorBrand = Symbol.for('kantha.gateway.error');

export class GatewayError extends Error {
  private readonly [gatewayErrorBrand] = true;

  // The Next.js instrumentation bundle and route bundles can each load this class.
  static [Symbol.hasInstance](value: unknown): boolean {
    return typeof value === 'object' && value !== null && gatewayErrorBrand in value;
  }

  public readonly code: GatewayErrorCode;
  public readonly requestId: string;
  public readonly retryAfterSeconds?: number;
  public readonly field?: string;
  public readonly resource?: string;
  public readonly details?: unknown;
  public readonly statusCode: number;

  constructor(params: {
    code: GatewayErrorCode;
    message: string;
    requestId: string;
    retryAfterSeconds?: number;
    field?: string;
    resource?: string;
    details?: unknown;
    statusCode?: number;
  }) {
    super(params.message);
    this.name = 'GatewayError';
    this.code = params.code;
    this.requestId = params.requestId;
    this.retryAfterSeconds = params.retryAfterSeconds;
    this.field = params.field;
    this.resource = params.resource;
    this.details = params.details;
    this.statusCode = params.statusCode ?? GATEWAY_HTTP_STATUS_MAP[params.code] ?? 500;
  }

  public toEnvelope(): GatewayErrorEnvelope {
    return {
      code: this.code,
      message: this.message,
      requestId: this.requestId,
      ...(this.retryAfterSeconds !== undefined ? { retryAfterSeconds: this.retryAfterSeconds } : {}),
      ...(this.field !== undefined ? { field: this.field } : {}),
      ...(this.resource !== undefined ? { resource: this.resource } : {}),
      ...(this.details !== undefined ? { details: this.details } : {}),
    };
  }
}

export function createGatewayError(params: {
  code: GatewayErrorCode;
  message: string;
  requestId?: string;
  retryAfterSeconds?: number;
  field?: string;
  resource?: string;
  details?: unknown;
  statusCode?: number;
}): GatewayError {
  return new GatewayError({
    code: params.code,
    message: params.message,
    requestId: params.requestId || `req_err_${Date.now().toString(36)}`,
    retryAfterSeconds: params.retryAfterSeconds,
    field: params.field,
    resource: params.resource,
    details: params.details,
    statusCode: params.statusCode,
  });
}

export function isGatewayErrorEnvelope(value: unknown): value is GatewayErrorEnvelope {
  if (!value || typeof value !== 'object') return false;
  const candidate = value as Record<string, unknown>;
  return (
    typeof candidate.code === 'string' &&
    typeof candidate.message === 'string' &&
    typeof candidate.requestId === 'string'
  );
}

/**
 * Gateway Runtime Validation & Schemas
 * Pure TypeScript runtime schema validators returning typed data or GatewayErrorEnvelope.
 */

import type { GatewayErrorEnvelope } from './errors';
import type { CreateConnectionInput, UpdateConnectionInput } from './connection';
import type { CreateSessionInput } from './session';
import type { GatewayChatRequest, UIMessage } from './chat';
import type { SubmitRunInput } from './run';

import type { GatewayEvent } from './event';

const PROVIDERS = ['openai', 'anthropic', 'groq', 'openrouter', 'ollama', 'deepseek', 'together', 'custom'];

export type ValidationResult<T> =
  | { ok: true; data: T }
  | { ok: false; error: GatewayErrorEnvelope };

function fail(field: string, message: string, requestId = 'req_validation'): { ok: false; error: GatewayErrorEnvelope } {
  return {
    ok: false,
    error: {
      code: 'invalid_request',
      message: `Validation failed: ${message}`,
      field,
      requestId,
    },
  };
}

export function validateCreateConnectionInput(raw: unknown, requestId?: string): ValidationResult<CreateConnectionInput> {
  if (!raw || typeof raw !== 'object') {
    return fail('body', 'Expected JSON object in request body', requestId);
  }
  const obj = raw as Record<string, unknown>;

  if (typeof obj.label !== 'string' || !obj.label.trim() || obj.label.trim().length > 100) {
    return fail('label', 'Label is required and must be a non-empty string', requestId);
  }

  if (typeof obj.endpoint !== 'string' || !obj.endpoint.trim()) {
    return fail('endpoint', 'Endpoint URL is required', requestId);
  }

  try {
    const parsedUrl = new URL(obj.endpoint.trim());
    if (!['http:', 'https:'].includes(parsedUrl.protocol)) {
      return fail('endpoint', 'Endpoint must have http or https protocol', requestId);
    }
  } catch {
    return fail('endpoint', 'Endpoint must be a valid URL', requestId);
  }

  if (typeof obj.providerType !== 'string' || !PROVIDERS.includes(obj.providerType)) {
    return fail('providerType', `Provider type must be one of: ${PROVIDERS.join(', ')}`, requestId);
  }

  if (obj.apiKey !== undefined && (typeof obj.apiKey !== 'string' || obj.apiKey.length > 4096 || /[\r\n]/.test(obj.apiKey))) {
    return fail('apiKey', 'Use a single-line API key of up to 4096 characters', requestId);
  }

  if (obj.eligibleModels !== undefined) {
    if (!Array.isArray(obj.eligibleModels) || !obj.eligibleModels.every((m) => typeof m === 'string')) {
      return fail('eligibleModels', 'eligibleModels must be an array of strings', requestId);
    }
  }

  return {
    ok: true,
    data: {
      label: obj.label.trim(),
      endpoint: obj.endpoint.trim(),
      providerType: obj.providerType as CreateConnectionInput['providerType'],
      apiKey: typeof obj.apiKey === 'string' ? obj.apiKey.trim() : undefined,
      eligibleModels: Array.isArray(obj.eligibleModels) ? obj.eligibleModels : undefined,
    },
  };
}

export function validateUpdateConnectionInput(raw: unknown, requestId?: string): ValidationResult<UpdateConnectionInput> {
  if (!raw || typeof raw !== 'object') {
    return fail('body', 'Expected JSON object in request body', requestId);
  }
  const obj = raw as Record<string, unknown>;

  const data: UpdateConnectionInput = {};

  if (obj.label !== undefined) {
    if (typeof obj.label !== 'string' || !obj.label.trim() || obj.label.trim().length > 100) {
      return fail('label', 'Label must be a non-empty string', requestId);
    }
    data.label = obj.label.trim();
  }

  if (obj.endpoint !== undefined) {
    if (typeof obj.endpoint !== 'string' || !obj.endpoint.trim()) {
      return fail('endpoint', 'Endpoint must be a non-empty string', requestId);
    }
    try {
      new URL(obj.endpoint.trim());
    } catch {
      return fail('endpoint', 'Endpoint must be a valid URL', requestId);
    }
    data.endpoint = obj.endpoint.trim();
  }

  if (obj.status !== undefined) {
    const validStatuses = ['active', 'inactive', 'testing', 'error', 'revoked'];
    if (typeof obj.status !== 'string' || !validStatuses.includes(obj.status)) {
      return fail('status', `Status must be one of: ${validStatuses.join(', ')}`, requestId);
    }
    data.status = obj.status as UpdateConnectionInput['status'];
  }

  if (obj.providerType !== undefined) {
    if (typeof obj.providerType !== 'string' || !PROVIDERS.includes(obj.providerType)) {
      return fail('providerType', 'Select a supported provider type', requestId);
    }
    data.providerType = obj.providerType as UpdateConnectionInput['providerType'];
  }

  if (obj.apiKey !== undefined) {
    if (typeof obj.apiKey !== 'string' || obj.apiKey.length > 4096 || /[\r\n]/.test(obj.apiKey)) {
      return fail('apiKey', 'Use a single-line API key of up to 4096 characters', requestId);
    }
    data.apiKey = obj.apiKey.trim();
  }

  if (obj.eligibleModels !== undefined) {
    if (!Array.isArray(obj.eligibleModels) || !obj.eligibleModels.every((m) => typeof m === 'string')) {
      return fail('eligibleModels', 'eligibleModels must be an array of strings', requestId);
    }
    data.eligibleModels = obj.eligibleModels;
  }

  return { ok: true, data };
}

export function validateUIMessage(raw: unknown, index: number, requestId?: string): ValidationResult<UIMessage> {
  if (!raw || typeof raw !== 'object') {
    return fail(`messages[${index}]`, 'Each message must be an object', requestId);
  }
  const msg = raw as Record<string, unknown>;

  if (typeof msg.id !== 'string' || !msg.id.trim()) {
    return fail(`messages[${index}].id`, 'Message id is required', requestId);
  }

  const validRoles = ['user', 'assistant', 'system'];
  if (typeof msg.role !== 'string' || !validRoles.includes(msg.role)) {
    return fail(`messages[${index}].role`, `Role must be one of: ${validRoles.join(', ')}`, requestId);
  }

  const parts: NonNullable<UIMessage['parts']> = [];
  if (Array.isArray(msg.parts)) {
    for (const item of msg.parts) {
      if (item && (item.type === 'text' || item.type === 'reasoning') && typeof item.text === 'string') {
        parts.push({ type: item.type, text: item.text });
      }
    }
  }
  const textParts = parts.filter((part) => part.type === 'text');
  if (!textParts.length && typeof msg.content !== 'string') {
    return fail(`messages[${index}].content`, 'Content must be a string', requestId);
  }

  return {
    ok: true,
    data: {
      id: msg.id,
      role: msg.role as UIMessage['role'],
      content: textParts.length ? textParts.map((part) => part.text).join('') : String(msg.content),
      parts: parts.length ? parts : undefined,
    },
  };
}

export function validateGatewayChatRequest(raw: unknown, requestId?: string): ValidationResult<GatewayChatRequest> {
  if (!raw || typeof raw !== 'object') {
    return fail('body', 'Expected JSON object in request body', requestId);
  }
  const obj = raw as Record<string, unknown>;

  if ('baseUrl' in obj || 'apiKey' in obj) {
    return fail('connectionId', 'Use a saved provider connection', requestId);
  }

  if (!Array.isArray(obj.messages) || obj.messages.length === 0 || obj.messages.length > 500) {
    return fail('messages', 'Messages array is required and cannot be empty', requestId);
  }

  const validatedMessages: UIMessage[] = [];
  for (let i = 0; i < obj.messages.length; i++) {
    const res = validateUIMessage(obj.messages[i], i, requestId);
    if (!res.ok) return res;
    validatedMessages.push(res.data);
  }

  return {
    ok: true,
    data: {
      messages: validatedMessages,
      model: typeof obj.model === 'string' ? obj.model : undefined,
      connectionId: typeof obj.connectionId === 'string' ? obj.connectionId : undefined,
      sessionId: typeof obj.sessionId === 'string' ? obj.sessionId : undefined,
      submissionKey: typeof obj.submissionKey === 'string' ? obj.submissionKey : undefined,
      system: typeof obj.system === 'string' ? obj.system : undefined,
      temperature: typeof obj.temperature === 'number' ? obj.temperature : undefined,
      maxTokens: typeof obj.maxTokens === 'number' ? obj.maxTokens : undefined,
      stream: typeof obj.stream === 'boolean' ? obj.stream : true,
    },
  };
}

    return fail('modelId', 'modelId is required', requestId);
  }
  if (!Array.isArray(obj.inputMessages) || obj.inputMessages.length === 0) {
    return fail('inputMessages', 'inputMessages array is required and cannot be empty', requestId);
  }

  const validatedMessages: UIMessage[] = [];
  for (let i = 0; i < obj.inputMessages.length; i++) {
    const res = validateUIMessage(obj.inputMessages[i], i, requestId);
    if (!res.ok) return res;
    validatedMessages.push(res.data);
  }

  return {
    ok: true,
    data: {
      sessionId: obj.sessionId.trim(),
      submissionKey: obj.submissionKey.trim(),
      modelId: obj.modelId.trim(),
      connectionId: typeof obj.connectionId === 'string' ? obj.connectionId.trim() : undefined,
      inputMessages: validatedMessages,
      system: typeof obj.system === 'string' ? obj.system : undefined,
      temperature: typeof obj.temperature === 'number' ? obj.temperature : undefined,
      maxTokens: typeof obj.maxTokens === 'number' ? obj.maxTokens : undefined,
    },
  };
}

import { createHash, randomUUID } from 'node:crypto';
import { APICallError, createUIMessageStreamResponse } from 'ai';
import type { LanguageModelUsage, ModelMessage, UIMessageChunk } from 'ai';
import type { UIMessage } from '../contracts/chat';
import type { RequestContext } from '../contracts/context';
import { GatewayError } from '../contracts/errors';
import type { GatewayRun } from '../contracts/run';
import { getGatewayRegistry } from '../server/registry';
import { gatewayModelPrice, quoteTokenCost } from './pricing';

const DEFAULT_MAX_OUTPUT_TOKENS = 2048;
const MAX_OUTPUT_TOKENS = 8192;
const MAX_ASSISTANT_TEXT_LENGTH = 256_000;
const RUN_HEADER = 'x-kantha-run-id';
const SESSION_HEADER = 'x-kantha-session-id';

function invalidChat(ctx: RequestContext, message: string, field?: string): never {
  throw new GatewayError({ code: 'invalid_request', message, field, requestId: ctx.requestId });
}

function readOptionalChatString(
  body: Record<string, unknown>,
  field: string,
  ctx: RequestContext,
  maxLength = 256
): string | undefined {
  const value = body[field];
  if (value === undefined) return undefined;
  if (typeof value !== 'string' || !value.trim() || value.length > maxLength || value.includes('\0')) {
    return invalidChat(ctx, `Invalid ${field}.`, field);
  }
  return value.trim();
}

function readTextMessage(value: unknown, fallbackId: string, ctx: RequestContext): UIMessage {
  if (!value || typeof value !== 'object' || Array.isArray(value)) {
    return invalidChat(ctx, 'Each chat message must be an object.', 'messages');
  }
  const message = value as Record<string, unknown>;
  if (message.role !== 'user' && message.role !== 'assistant') {
    return invalidChat(ctx, 'Chat accepts user and assistant text messages.', 'messages');
  }

  let content: string;
  if (Array.isArray(message.parts)) {
    const chunks: string[] = [];
    for (const part of message.parts) {
      if (part?.type === 'file') {
        return invalidChat(ctx, 'Provider chat currently accepts text only.', 'messages');
      }
      if (part?.type === 'text' && typeof part.text === 'string') chunks.push(part.text);
    }
    content = chunks.join('');
  } else {
    content = typeof message.content === 'string' ? message.content : typeof message.text === 'string' ? message.text : '';
  }

  if (message.role === 'user' && !content.trim()) {
    return invalidChat(ctx, 'A user message must contain text.', 'messages');
  }
  if (message.id !== undefined && (typeof message.id !== 'string' || !message.id.trim() || message.id.length > 256)) {
    return invalidChat(ctx, 'Invalid message ID.', 'messages');
  }

  return {
    id: typeof message.id === 'string' ? message.id : fallbackId,
    role: message.role,
    content,
    parts: [{ type: 'text', text: content }],
  };
}

function readChatInput(raw: unknown, ctx: RequestContext, sessionPath?: string) {
  if (!raw || typeof raw !== 'object' || Array.isArray(raw)) {
    return invalidChat(ctx, 'A JSON chat request is required.');
  }
  const body = raw as Record<string, unknown>;
  if ('baseUrl' in body || 'apiKey' in body || 'endpoint' in body) {
    return invalidChat(ctx, 'Use a configured provider connection for chat.', 'connectionId');
  }

  const sessionId = sessionPath ?? readOptionalChatString(body, 'sessionId', ctx);
  const submissionKey = readOptionalChatString(body, 'submissionKey', ctx);
  const modelId = readOptionalChatString(body, sessionPath ? 'modelId' : 'model', ctx);
  const connectionId = readOptionalChatString(body, 'connectionId', ctx);
  const system = readOptionalChatString(body, 'system', ctx, 8192);
  const temperature = body.temperature;
  const maxTokens = body.maxTokens ?? DEFAULT_MAX_OUTPUT_TOKENS;
  if (
    typeof maxTokens !== 'number' ||
    !Number.isInteger(maxTokens) ||
    maxTokens < 1 ||
    maxTokens > MAX_OUTPUT_TOKENS
  ) {
    return invalidChat(ctx, `maxTokens must be between 1 and ${MAX_OUTPUT_TOKENS}.`, 'maxTokens');
  }
  if (
    temperature !== undefined &&
    (typeof temperature !== 'number' || !Number.isFinite(temperature) || temperature < 0 || temperature > 2)
  ) {
    return invalidChat(ctx, 'temperature must be between 0 and 2.', 'temperature');
  }

  const rawMessages: unknown = sessionPath ? body.inputMessages : body.messages;
  if (!Array.isArray(rawMessages) || rawMessages.length === 0 || rawMessages.length > 500) {
    return invalidChat(ctx, 'Supply between 1 and 500 chat messages.', 'messages');
  }
  if (sessionPath && (!submissionKey || !modelId)) {
    return invalidChat(ctx, 'A submission key and model ID are required for this run.');
  }

  // Existing sessions take their history from the journal. Only the new user turn comes from the client.
  const incoming = sessionId ? rawMessages.slice(-1) : rawMessages;
  const fallbackId = submissionKey
    ? `message_${createHash('sha256').update(submissionKey).digest('hex').slice(0, 32)}`
    : `message_${randomUUID()}`;
  const inputMessages = incoming.map((value, index) =>
    readTextMessage(value, `${fallbackId}_${index}`, ctx)
  );
  const lastMessage = inputMessages[inputMessages.length - 1];
  if (lastMessage.role !== 'user') {
    return invalidChat(ctx, 'The last chat message must be from the user.', 'messages');
  }

  return {
    connectionId,
    modelId,
    sessionId,
    submissionKey: submissionKey ?? lastMessage.id,
    inputMessages,
    system,
    temperature,
    maxTokens,
  };
}

function providerError(error: unknown, ctx: RequestContext, timedOut: boolean): GatewayError {
  if (timedOut || (error instanceof Error && error.name === 'TimeoutError')) {
    return new GatewayError({
      code: 'upstream_timeout',
      message: 'The provider took too long to respond.',
      requestId: ctx.requestId,
    });
  }
  if (APICallError.isInstance(error) && error.statusCode === 429) {
    const retryAfter = error.responseHeaders?.['retry-after'];
    const seconds = retryAfter && /^\d+$/.test(retryAfter) ? Number(retryAfter) : undefined;
    return new GatewayError({
      code: 'rate_limited',
      message: 'The provider is rate-limiting requests. Try again later.',
      retryAfterSeconds: seconds !== undefined && Number.isSafeInteger(seconds) ? seconds : undefined,
      requestId: ctx.requestId,
    });
  }
  const message = APICallError.isInstance(error) && (error.statusCode === 401 || error.statusCode === 403)
    ? 'The provider rejected this connection. Ask an owner to check its credentials.'
    : 'The provider could not complete the response.';
  return new GatewayError({ code: 'upstream_failed', message, requestId: ctx.requestId });
}

function reportedTokens(usage: LanguageModelUsage | undefined) {
  if (usage?.inputTokens === undefined || usage.outputTokens === undefined) return undefined;
  const { inputTokens: prompt, outputTokens: completion, totalTokens = prompt + completion } = usage;
  if (![prompt, completion, totalTokens].every((value) => Number.isSafeInteger(value) && value >= 0)) {
    return undefined;
  }
  return { prompt, completion, total: Math.max(totalTokens, prompt + completion) };
}

function chatStreamHeaders(run: GatewayRun) {
  return {
    [RUN_HEADER]: run.id,
    [SESSION_HEADER]: run.sessionId,
    'cache-control': 'no-store, no-transform',
    'x-accel-buffering': 'no',
  };
}

function replayChatRun(run: GatewayRun) {
  return {
    run,
    replayed: true,
    response: createUIMessageStreamResponse({
      headers: { ...chatStreamHeaders(run), 'x-kantha-replayed': 'true' },
      stream: new ReadableStream<UIMessageChunk>({
        start(controller) {
          controller.close();
        },
      }),
    }),
  };
}

/** Starts one durable run and returns its standard AI SDK browser stream. */
export async function startGatewayChat(raw: unknown, ctx: RequestContext, sessionPath?: string) {
  const input = readChatInput(raw, ctx, sessionPath);
  const registry = getGatewayRegistry();
  let { connectionId, modelId } = input;
  if (!connectionId || !modelId) {
    const connections = await registry.connectionStore.listConnections(ctx);
    const selected = connections.find((connection) =>
      connection.status === 'active' &&
      (!connectionId || connection.id === connectionId) &&
      (modelId ? connection.eligibleModels.includes(modelId) : connection.eligibleModels.length > 0)
    );
    connectionId = selected?.id;
    modelId = modelId ?? selected?.eligibleModels[0];
  }
  if (!connectionId || !modelId) {
    return invalidChat(ctx, 'Select an eligible model from an active provider connection.', 'model');
  }

  const sessionId = input.sessionId ?? (await registry.runService.createSession({}, ctx)).id;
  const submission = { ...input, sessionId, connectionId, modelId };
  const previous = await registry.runService.getRunBySubmission(submission, ctx);
  if (previous) return replayChatRun(previous);

  const connection = await registry.connectionStore.getConnection(connectionId, ctx);
  if (!connection) {
    throw new GatewayError({ code: 'not_found', message: 'Provider connection was not found.', requestId: ctx.requestId });
  }
  if (connection.status !== 'active' || !connection.eligibleModels.includes(modelId)) {
    return invalidChat(ctx, 'The selected model is unavailable on this connection.', 'model');
  }

  const secret = connection.secretRef
    ? await registry.secretStore.getSecret(connection.secretRef, ctx)
    : null;
  if (connection.secretRef && !secret) {
    throw new GatewayError({ code: 'conflict', message: 'Provider credentials changed. Refresh the connection and try again.', requestId: ctx.requestId });
  }
  const price = gatewayModelPrice(connection, modelId, ctx);

  const { run, created } = await registry.runService.beginRun(submission, ctx);
  if (!created) return replayChatRun(run);

  const messageId = `assistant_${run.id}`;
  const startedAt = Date.now();
  const attemptId = randomUUID();
  const reservationId = randomUUID();
  const timeout = AbortSignal.timeout(55_000);
  const outputLimit = new AbortController();
  const runSignal = registry.runService.getRunSignal(run.id, ctx);
  const signal = AbortSignal.any([runSignal, timeout, outputLimit.signal]);
  let networkDispatched = false;
  let reserved = false;
  let outputLength = 0;
  let finishing: Promise<void> | undefined;

  function finish(
    status: 'completed' | 'failed' | 'cancelled',
    details?: { text?: string; error?: GatewayError; usage?: LanguageModelUsage }
  ) {
    finishing ??= (async () => {
      if (reserved) {
        const tokens = reportedTokens(details?.usage);
        await registry.usageService.settleAttempt({
          attemptId,
          reservationId,
          workspaceId: ctx.workspaceId,
          connectionId: run.connectionId,
          modelId: run.modelId,
          runId: run.id,
          status: networkDispatched ? 'settled' : 'refunded',
          tokens,
          costUsd: tokens ? quoteTokenCost(price, tokens.prompt, tokens.completion) : undefined,
          durationMs: Date.now() - startedAt,
        }, ctx);
      }
      const message: UIMessage | undefined = details?.text
        ? {
            id: messageId,
            role: 'assistant',
            content: details.text,
            parts: [{ type: 'text', text: details.text }],
          }
        : undefined;
      await registry.runService.finishRun(run.id, { status, message, error: details?.error?.toEnvelope() }, ctx);
    })();
    return finishing;
  }
  const finishError = (error: unknown) => runSignal.aborted
    ? finish('cancelled')
    : finish('failed', { error: providerError(error, ctx, timeout.aborted) });

  try {
    const snapshot = await registry.runService.getSnapshot(sessionId, ctx);
    const messages: ModelMessage[] = snapshot.messages.flatMap((message): ModelMessage[] =>
      message.content && (message.role === 'user' || message.role === 'assistant')
        ? [{ role: message.role, content: message.content }]
        : []
    );
    let estimatedInputTokens = 32 + Buffer.byteLength(input.system ?? '');
    for (const message of snapshot.messages) {
      estimatedInputTokens += 16 + Buffer.byteLength(message.content);
    }
    reserved = await registry.usageService.reserveAttempt({
      attemptId,
      reservationId,
      workspaceId: ctx.workspaceId,
      connectionId,
      modelId,
      runId: run.id,
      status: 'reserved',
      retryCount: 0,
      estimatedTokens: estimatedInputTokens + input.maxTokens,
      estimatedCostUsd: quoteTokenCost(price, estimatedInputTokens, input.maxTokens),
    }, ctx);
    if (!reserved) {
      throw new GatewayError({
        code: 'budget_exceeded',
        message: 'The workspace budget cannot admit this request.',
        requestId: ctx.requestId,
      });
    }
    signal.throwIfAborted();

    const result = registry.inferenceAdapter.streamChat({
      modelId,
      messages,
      system: input.system,
      temperature: input.temperature,
      maxTokens: input.maxTokens,
      signal,
      onDispatch: () => { networkDispatched = true; },
      onText: async (delta) => {
        outputLength += delta.length;
        if (outputLength > MAX_ASSISTANT_TEXT_LENGTH) {
          outputLimit.abort();

          await finishError(signal.reason);
          return;
        }
        await finish('completed', { text, usage });
      },
      onError: finishError,
      onAbort: () => finishError(signal.reason),
    }, connection, secret ?? undefined);

    const response = result.toUIMessageStreamResponse({
      headers: chatStreamHeaders(run),
      generateMessageId: () => messageId,
      messageMetadata: ({ part }) => part.type === 'start' || part.type === 'finish'
        ? { runId: run.id, sessionId }
        : undefined,
      sendReasoning: false,
      onError: (error) => providerError(error, ctx, timeout.aborted).message,
      consumeSseStream: async ({ stream }) => {
        // Keep writing the journal when the browser disconnects. Explicit Stop cancels by run ID.
        try {
          await stream.pipeTo(new WritableStream<string>());
          if (!finishing) await finishError(new Error('Provider stream closed without a terminal frame'));
        } catch (error) {
          try {
            await finishError(error);
          } catch {
            console.error('Gateway could not finalize run', ctx.requestId);
          }
        }
      },
    });

    return { run, response, replayed: false };
  } catch (error) {
    const failure = error instanceof GatewayError ? error : providerError(error, ctx, timeout.aborted);
    if (runSignal.aborted) await finish('cancelled');
    else await finish('failed', { error: failure });

  }

function isAllowedPrivateOrigin(url: URL): boolean {

function checkUrl(url: URL) {

export function normalizeProviderEndpoint(endpoint: string): string {
  if (endpoint.length > 2048) throw endpointError();
  let url: URL;
  try {
    url = new URL(endpoint);
  } catch {
    throw endpointError();
  }
  checkUrl(url);
  if (url.search) throw endpointError();

  url.pathname = url.pathname.replace(/\/+$/, '').replace(/\/(?:models|chat\/completions)$/, '');

async function resolveAddress(url: URL, signal: AbortSignal): Promise<string> {
  const hostname = url.hostname.replace(/\.$/, '');
  if (isIP(hostname) === 4) return hostname;
  const addresses = await lookup(hostname, { family: 4, all: true });
  signal.throwIfAborted();
  if (!addresses.length || (!isAllowedPrivateOrigin(url) && addresses.some(({ address }) => DISALLOWED.check(address)))) {
    throw endpointError();
  }
  return addresses[0].address;
}

export const providerFetch: typeof fetch = async (input, init) => {
  const outbound = new Request(input, init);
  const url = new URL(outbound.url);
  checkUrl(url);
  const signal = AbortSignal.any([outbound.signal, AbortSignal.timeout(60_000)]);
  const address = await resolveAddress(url, signal);
  const body = outbound.body ? Buffer.from(await outbound.arrayBuffer()) : undefined;
  const headers = Object.fromEntries(outbound.headers);
  headers['accept-encoding'] = 'identity';
  delete headers.host;
  delete headers.cookie;
  if (body) headers['content-length'] = String(body.byteLength);
  const send = url.protocol === 'https:' ? requestHttps : requestHttp;

  return new Promise<Response>((resolve, reject) => {
    const request = send(url, {
      method: outbound.method,
      headers,
      agent: false,
      family: 4,
      lookup: (_hostname, _options, callback) => callback(null, address, 4),
      signal,
    }, (incoming) => {
      const status = incoming.statusCode ?? 502;
      if (status >= 300 && status < 400) {
        incoming.destroy();
        reject(createGatewayError({ code: 'upstream_failed', message: 'Provider redirects are disabled.' }));
        return;
      }

      const responseHeaders = new Headers();
      for (const [name, value] of Object.entries(incoming.headers)) {
        for (const item of Array.isArray(value) ? value : [value]) {
          if (item !== undefined) responseHeaders.append(name, item);
        }
      }
      const hasBody = outbound.method !== 'HEAD' && status !== 204 && status !== 205;
      if (!hasBody) {
        incoming.resume();
        resolve(new Response(null, { status, headers: responseHeaders }));
        return;
      }
      const encoding = responseHeaders.get('content-encoding');
      if ((encoding && encoding !== 'identity') ||
          Number(responseHeaders.get('content-length')) > MAX_UPSTREAM_BYTES) {
        incoming.destroy();
        reject(createGatewayError({ code: 'upstream_failed', message: 'Provider response exceeds transport limits.' }));
        return;
      }
      let received = 0;
      const stream = Readable.toWeb(incoming, {
        strategy: { highWaterMark: incoming.readableHighWaterMark, size: (chunk: Uint8Array) => chunk.byteLength },
      }) as ReadableStream<Uint8Array>;
      const body = stream.pipeThrough(new TransformStream<Uint8Array, Uint8Array>({
        transform(chunk, controller) {
          received += chunk.byteLength;
          if (received > MAX_UPSTREAM_BYTES) {
            throw createGatewayError({ code: 'upstream_failed', message: 'Provider response exceeds transport limits.' });
          }
          controller.enqueue(chunk);
        },
      }));
      resolve(new Response(body, { status, headers: responseHeaders }));
    });
    request.once('error', reject);
    request.end(body);
  });
};

export function makeProviderFetch(endpoint: string): typeof fetch {
  const base = new URL(normalizeProviderEndpoint(endpoint));
  const prefix = base.pathname.replace(/\/+$/, '');
  return (input, init) => {
    const url = new URL(input instanceof Request ? input.url : input);
    if (url.origin !== base.origin ||
        (url.pathname !== prefix && !url.pathname.startsWith(prefix + '/'))) {
      throw endpointError();
    }
    return providerFetch(input, init);
  };
}

export function gatewayModelPrice(
  connection: GatewayConnection,
  modelId: string,
  ctx: RequestContext
): TokenPrice | undefined {
  const configured = process.env.GATEWAY_MODEL_PRICES;
  if (!configured?.trim()) return undefined;

  const invalidPrices = () => new GatewayError({
    code: 'service_unavailable',
    message: 'Gateway pricing configuration is invalid.',
    requestId: ctx.requestId,
  });

  let raw: unknown;
  try {
    raw = JSON.parse(configured);
  } catch {
    throw invalidPrices();
  }
  if (!raw || typeof raw !== 'object' || Array.isArray(raw)) throw invalidPrices();

  const prices = new Map<string, TokenPrice>();
  for (const [key, value] of Object.entries(raw as Record<string, unknown>)) {
    if (!value || typeof value !== 'object' || Array.isArray(value)) throw invalidPrices();
    const { inputPerMillion, outputPerMillion } = value as Record<string, unknown>;
    const separator = key.indexOf(':');
    if (separator < 1 || separator === key.length - 1 || !validRate(inputPerMillion) || !validRate(outputPerMillion)) {
      throw invalidPrices();
    }
    prices.set(key, { inputPerMillion, outputPerMillion });
  }

  return prices.get(`${connection.id}:${modelId}`) ?? prices.get(`${connection.providerType}:${modelId}`);
}

  price: TokenPrice | undefined,
  inputTokens: number,
  outputTokens: number

  if (!price) return undefined;
  return (price.inputPerMillion * inputTokens + price.outputPerMillion * outputTokens) / 1_000_000;

      const record = this.refreshActiveRun(stored, ctx);
      if (record.session.activeRunId) {
        throw new GatewayError({ code: 'conflict', message: 'Cancel the active run before deleting this session', requestId: ctx.requestId, resource: id });
      }
      record.session.status = 'deleted';
      record.session.updatedAt = new Date().toISOString();
      record.messages = [];
      this.db.deleteSessionEvents(ctx.workspaceId, id);
      this.db.put('sessions', ctx.workspaceId, id, record);
      return true;
    });
  }

  private findSubmission(input: SubmitRunInput, ctx: RequestContext): GatewayRun | null {
    const key = JSON.stringify([input.sessionId, input.submissionKey]);
    const runId = this.db.get<string>('submissions', ctx.workspaceId, key);
    if (!runId) return null;
    const stored = this.db.get<RunRecord>('runs', ctx.workspaceId, runId);
    if (!stored) throw new Error('The stored submission has no associated run.');
    if (stored.payloadDigest !== digest(input)) {
      throw new GatewayError({ code: 'conflict', message: 'This submission key belongs to a different request', requestId: ctx.requestId });
    }
    return stored.run;
  }

  async getRunBySubmission(input: SubmitRunInput, ctx: RequestContext): Promise<GatewayRun | null> {
    return this.db.write(() => {
      this.refreshActiveRun(this.requireSession(input.sessionId, ctx), ctx);
      return this.findSubmission(input, ctx);
    });
  }

  async beginRun(input: SubmitRunInput, ctx: RequestContext): Promise<{ run: GatewayRun; created: boolean }> {
    return this.db.write(() => {
      const session = this.refreshActiveRun(this.requireSession(input.sessionId, ctx), ctx);
      const existing = this.findSubmission(input, ctx);
      if (existing) return { run: existing, created: false };
      if (session.session.activeRunId || session.session.status !== 'active') {
        throw new GatewayError({ code: 'conflict', message: 'This session cannot start another run yet', requestId: ctx.requestId, resource: input.sessionId });
      }

      const known = new Map(session.messages.map((message) => [message.id, message]));
      const messageCount = known.size;
      for (const message of input.inputMessages) {
        const previous = known.get(message.id);
        if (previous && (previous.role !== message.role || previous.content !== message.content)) {
          throw new GatewayError({ code: 'conflict', message: 'This message ID already has different content', requestId: ctx.requestId });
        }
        if (!previous) known.set(message.id, message);
      }
      if (known.size === messageCount) {
        throw new GatewayError({ code: 'conflict', message: 'This run needs a new message', requestId: ctx.requestId });
      }

      const run: GatewayRun = {
        id: `run_${randomUUID()}`,
        sessionId: input.sessionId,
        workspaceId: ctx.workspaceId,
        submissionKey: input.submissionKey,
        status: 'running',
        modelId: input.modelId,
        connectionId: input.connectionId ?? '',
        startedAt: new Date().toISOString(),
      };
      const stored: RunRecord = { run, payloadDigest: digest(input), leaseUntil: Date.now() + runLeaseMs };
      this.db.put('runs', ctx.workspaceId, run.id, stored);
      this.db.put('submissions', ctx.workspaceId, JSON.stringify([input.sessionId, input.submissionKey]), run.id);
      session.session.activeRunId = run.id;
      session.latestRunId = run.id;
      session.messages = [...known.values()];
      recordSessionEvent(this.db, session, 'run.state_changed', { previousState: 'queued', newState: 'running' }, run.id);
      return { run, created: true };
    });
  }

  async submitRun(input: SubmitRunInput, ctx: RequestContext): Promise<GatewayRun> {
    return (await this.beginRun(input, ctx)).run;
  }

  async getRun(id: string, ctx: RequestContext): Promise<GatewayRun | null> {
    return this.db.write(() => {
      const stored = this.db.get<RunRecord>('runs', ctx.workspaceId, id);
      if (!stored) return null;

  async appendAssistantDelta(runId: string, messageId: string, delta: string, ctx: RequestContext): Promise<void> {
    if (!delta) return;
    this.db.write(() => {
      const stored = this.db.get<RunRecord>('runs', ctx.workspaceId, runId);
      if (!stored || stored.run.status !== 'running') return;
      const session = this.requireSession(stored.run.sessionId, ctx);
      if (session.session.activeRunId !== runId) return;
      const previous = session.messages.find((message) => message.id === messageId);
      const content = (previous?.content ?? '') + delta;
      session.messages = mergeMessages(session.messages, [{
        id: messageId, role: 'assistant', content, parts: [{ type: 'text', text: content }],
      }]);
      recordSessionEvent(this.db, session, 'run.delta', { delta, role: 'assistant', messageId }, runId);
    });
  }

  async finishRun(id: string, input: EndRunInput, ctx: RequestContext): Promise<GatewayRun> {
    const run = this.db.write(() => {
      const stored = this.db.get<RunRecord>('runs', ctx.workspaceId, id);
      if (!stored) throw new GatewayError({ code: 'not_found', message: 'Run not found', requestId: ctx.requestId, resource: id });
      const session = this.requireSession(stored.run.sessionId, ctx);
      if (!isActive(stored.run.status)) return stored.run;
      if (stored.run.pendingApprovalId) {
        const approval = this.db.get<GatewayApproval>('approvals', ctx.workspaceId, stored.run.pendingApprovalId);
        if (approval) {
          approval.status = 'interrupted';
          approval.actorId = ctx.caller.id;
          approval.resolvedAt = new Date().toISOString();
          this.db.put('approvals', ctx.workspaceId, approval.id, approval);
          recordSessionEvent(this.db, session, 'approval.resolved', { approvalId: approval.id, decision: 'interrupted' }, id);
        }
      }
      const messages = input.message ? [input.message] : (input.messages ?? []);
      session.messages = mergeMessages(session.messages, messages);
      stored.run.error = input.error;
      endRun(this.db, session, stored, input.status, input.reason ?? input.error?.message);
      return stored.run;
    });
    this.stopSignal(ctx.workspaceId, id, run.status === 'cancelled' || run.status === 'interrupted');
    return run;
  }

  async cancelRun(id: string, reason: string | undefined, ctx: RequestContext): Promise<GatewayRun> {
    return this.finishRun(id, { status: 'cancelled', reason }, ctx);
  }

  getRunSignal(id: string, ctx: RequestContext, requestSignal?: AbortSignal): AbortSignal {
    const key = JSON.stringify([ctx.workspaceId, id]);
    const existing = this.signals.get(key);
    if (existing) return requestSignal ? AbortSignal.any([requestSignal, existing.controller.signal]) : existing.controller.signal;

    const controller = new AbortController();
    const checkRun = () => {
      try {
        const stored = this.db.get<RunRecord>('runs', ctx.workspaceId, id);
        if (stored?.run.status !== 'running' || !stored.leaseUntil) {
          this.stopSignal(ctx.workspaceId, id, true);
          return;
        }
        if (stored.leaseUntil - Date.now() < runLeaseMs - 5_000) {
          const renewed = this.db.write(() => {
            const fresh = this.db.get<RunRecord>('runs', ctx.workspaceId, id);
            if (fresh?.run.status !== 'running' || !fresh.leaseUntil || fresh.leaseUntil <= Date.now()) return false;
            fresh.leaseUntil = Date.now() + runLeaseMs;
            this.db.put('runs', ctx.workspaceId, id, fresh);
            return true;
          });
          if (!renewed) this.stopSignal(ctx.workspaceId, id, true);
        }
      } catch {
        this.stopSignal(ctx.workspaceId, id, true);
      }
    };
    const poll = setInterval(checkRun, 250);
    poll.unref();
    this.signals.set(key, { controller, poll });
    checkRun();
    return requestSignal ? AbortSignal.any([requestSignal, controller.signal]) : controller.signal;
  }

  private stopSignal(scope: string, id: string, abort: boolean): void {
    const key = JSON.stringify([scope, id]);
    const signal = this.signals.get(key);
    if (signal) {
      clearInterval(signal.poll);
      this.signals.delete(key);
      if (abort) signal.controller.abort();
    }
  }

  shutdown(): void {
    for (const signal of this.signals.values()) {
      clearInterval(signal.poll);
      signal.controller.abort();
    }
    this.signals.clear();
  }

  async createInertApproval(sessionId: string, ctx: RequestContext): Promise<GatewayApproval> {
    return this.db.write(() => {

        throw new GatewayError({ code: 'conflict', message: 'Finish the active run before starting an approval test', requestId: ctx.requestId, resource: sessionId });

      const approval: GatewayApproval = {
        id: `approval_${randomUUID()}`,
        runId,
        sessionId,
        workspaceId: ctx.workspaceId,
        actionVersion: 1,
        toolName: 'inert_fixture_tool',
        summary: 'Run the gateway approval self-test',
        parameters: { target: 'gateway-self-test' },
        status: 'waiting',
        createdAt: now.toISOString(),

      };
      const run: GatewayRun = {
        id: runId,
        sessionId,
        workspaceId: ctx.workspaceId,
        submissionKey: `inert_${randomUUID()}`,
        status: 'waiting_for_approval',
        modelId: '',
        connectionId: '',
        startedAt: now.toISOString(),

const SESSION_LIFETIME = 8 * 60 * 60;

  workspaceId: string;

function configurationError() {
  return createGatewayError({
    code: 'service_unavailable',
    message: 'Gateway access is not configured.',
  });

function digest(token: string): string {
  return createHash('sha256').update(token).digest('hex');
}

function members(): WorkspaceMember[] {
  const configured = process.env.GATEWAY_MEMBERS;
  if (configured) {
    let entries: WorkspaceMember[];
    try {
      entries = JSON.parse(configured);
    } catch {
      throw configurationError();
    }

    if (!Array.isArray(entries) || !entries.length || entries.length > 200) {
      throw configurationError();
    }

    for (const entry of entries) {
      if (!entry ||
        typeof entry.actorId !== 'string' || !ID_PATTERN.test(entry.actorId) ||
        typeof entry.workspaceId !== 'string' || !ID_PATTERN.test(entry.workspaceId) ||
        (entry.role !== 'owner' && entry.role !== 'member') ||
        typeof entry.tokenSha256 !== 'string' || !/^[0-9a-f]{64}$/.test(entry.tokenSha256)) {
        throw configurationError();
      }
    }

    if (new Set(entries.map((entry) => entry.tokenSha256)).size !== entries.length) {
      throw configurationError();
    }
    return entries;
  }

  const token =
    process.env.GATEWAY_ACCESS_TOKEN ||
    (process.env.NODE_ENV !== 'production' ? 'kantha-dev-workspace-access-token-32chars' : undefined);
  const workspaceId = process.env.GATEWAY_WORKSPACE_ID ?? 'kantha';
  if (!token || !TOKEN_PATTERN.test(token) || !ID_PATTERN.test(workspaceId)) {
    throw configurationError();
  }

  return [{ actorId: 'owner', workspaceId, role: 'owner', tokenSha256: digest(token) }];
}

function equalHex(first: string, second: string): boolean {
  return timingSafeEqual(Buffer.from(first, 'hex'), Buffer.from(second, 'hex'));
}

export function checkGatewayAccessConfiguration(): void {
  members();
}

function memberForToken(token: string, available: WorkspaceMember[]): WorkspaceMember | undefined {
  if (!TOKEN_PATTERN.test(token)) return;
  const tokenHash = digest(token);
  return available.find((member) => equalHex(member.tokenSha256, tokenHash));
}

function signature(member: WorkspaceMember, text: string): string {
  return createHmac('sha256', Buffer.from(member.tokenSha256, 'hex'))
    .update(JSON.stringify(['kantha-access-v1', member.actorId, member.workspaceId, member.role, text]))
    .digest('hex');
}

function memberForCookie(cookie: string, available: WorkspaceMember[]): WorkspaceMember | undefined {
  const [expires, nonce, proof, extra] = cookie.split('.');
  if (extra !== undefined ||
      !/^\d{10}$/.test(expires ?? '') ||
      !/^[a-f0-9-]{36}$/.test(nonce ?? '') ||
      !/^[a-f0-9]{64}$/.test(proof ?? '') ||
      Number(expires) <= Math.floor(Date.now() / 1000)) {
    return;
  }
  return available.find((member) => equalHex(signature(member, [expires, nonce].join('.')), proof));
}

export function gatewayOrigin(request: Request): URL {
  const origin = new URL(process.env.GATEWAY_ORIGIN || request.url);
  if (process.env.NODE_ENV === 'production' && origin.protocol !== 'https:' &&
      !['localhost', '127.0.0.1', '[::1]'].includes(origin.hostname)) {
    throw configurationError();
  }
  return origin;
}

export function requireSameOrigin(request: Request): void {
  const expected = gatewayOrigin(request).origin;
  if (request.headers.get('origin') !== expected) {
    throw createGatewayError({ code: 'forbidden', message: 'This request must come from the gateway origin.' });
  }
}

export function authenticateGatewayRequest(request: NextRequest): WorkspaceMember | undefined {
  const credentials = members();
  const authorization = request.headers.get('authorization');
  if (authorization) {
    const match = /^Bearer ([A-Za-z0-9_-]{32,256})$/i.exec(authorization);
    return match ? memberForToken(match[1], credentials) : undefined;
  }
  const cookie = request.cookies.get(ACCESS_COOKIE)?.value;
  if (!cookie) return;
  if (request.method !== 'GET' && request.method !== 'HEAD') requireSameOrigin(request);
  return memberForCookie(cookie, credentials);
}

export function grantGatewayAccess(token: string) {
  const member = memberForToken(token, members());
  if (!member) throw createGatewayError({ code: 'unauthorized', message: 'Access token is invalid.' });
  const text = [Math.floor(Date.now() / 1000) + SESSION_LIFETIME, randomUUID()].join('.');
  return { member, cookie: [text, signature(member, text)].join('.'), maxAge: SESSION_LIFETIME };
}

export function requireGatewayOwner(ctx: RequestContext): void {
  if (!ctx.caller.roles?.includes('owner')) {
    throw createGatewayError({ code: 'forbidden', message: 'Workspace owner access is required.', requestId: ctx.requestId });
  }

        throw new GatewayError({ code: 'conflict', message: 'Connection changed. Refresh and try again.', requestId: ctx.requestId, resource: id });

        (input.endpoint !== undefined && input.endpoint !== existing.endpoint) ||
        (input.providerType !== undefined && input.providerType !== existing.providerType);

        secretRef = `sec_${randomUUID()}`;
        this.secrets.writeSecret(secretRef, input.apiKey, ctx);
      } else if (secretRef && (revoked || endpointChanged || input.apiKey === '')) {
        this.secrets.removeSecret(secretRef, ctx);
        secretRef = undefined;
      }

      const updated: GatewayConnection = {
        ...existing,
        label: input.label ?? existing.label,
        endpoint: input.endpoint ?? existing.endpoint,
        providerType: input.providerType ?? existing.providerType,
        status: input.status ?? existing.status,
        eligibleModels: revoked || endpointChanged || credentialsChanged ? [] : (input.eligibleModels ?? existing.eligibleModels),
        lastTestedAt: endpointChanged || credentialsChanged ? undefined : existing.lastTestedAt,
        lastError: endpointChanged || credentialsChanged ? undefined : existing.lastError,
        secretRef,
        updatedAt: new Date().toISOString(),
        revision: randomUUID(),
      };
      this.db.put('connections', ctx.workspaceId, id, updated);
      return updated;
    });
  }

    outcome: { testedAt: string; success: boolean },

      if (current.lastTestedAt && current.lastTestedAt > outcome.testedAt) return current;

      current.lastError = outcome.success ? undefined : 'Provider connection test failed';
      this.db.put('connections', ctx.workspaceId, id, current, Date.parse(current.updatedAt));
      return current;
    });
  }

export async function readGatewayJson(req: Request, maxBytes = 1024 * 1024): Promise<unknown> {
  const bodyType = req.headers.get('content-type');
  if (bodyType && bodyType.split(';', 1)[0].trim().toLowerCase() !== 'application/json') {
    throw createGatewayError({ code: 'invalid_request', message: 'Expected an application/json body.', statusCode: 415 });
  }
  if (Number(req.headers.get('content-length')) > maxBytes) {
    throw createGatewayError({ code: 'invalid_request', message: 'Request body is too large.', statusCode: 413 });
  }
  if (!req.body) return null;

  const reader = req.body.getReader();
  const decoder = new TextDecoder('utf-8', { fatal: true });
  let size = 0;
  let body = '';
  try {
    for (;;) {
      const { done, value } = await reader.read();
      if (done) break;
      size += value.byteLength;
      if (size > maxBytes) {
        await reader.cancel();
        throw createGatewayError({ code: 'invalid_request', message: 'Request body is too large.', statusCode: 413 });
      }
      body += decoder.decode(value, { stream: true });
    }
    body += decoder.decode();
    return body.trim() ? JSON.parse(body) : null;
  } catch (error) {
    if (error instanceof GatewayError) throw error;
    throw createGatewayError({ code: 'invalid_request', message: 'Request body contains invalid JSON.' });
  } finally {
    reader.releaseLock();
  }
}

export function formatErrorResponse(
  error: unknown,
  requestId: string,
): { response: NextResponse; envelope: GatewayErrorEnvelope } {
  const isGwError =
    error instanceof GatewayError ||
    (typeof error === 'object' &&
      error !== null &&
      ('code' in error || (error as { name?: string }).name === 'GatewayError'));
  const failure = isGwError
    ? (error as GatewayError)
    : createGatewayError({ code: 'internal_error', message: 'An internal gateway error occurred.' });
  if (!isGwError) {
    console.error('Gateway request failed', error);
  }
  const { code, message, field, retryAfterSeconds } = failure;
  const envelope: GatewayErrorEnvelope = {
    code,
    message,
    requestId,
    ...(field ? { field } : {}),
    ...(retryAfterSeconds !== undefined ? { retryAfterSeconds } : {}),
  };
  const headers: Record<string, string> = {
    [GATEWAY_HEADERS.REQUEST_ID]: requestId,
    'cache-control': 'no-store',
  };
  if (retryAfterSeconds !== undefined) headers['retry-after'] = String(retryAfterSeconds);
  return {
    response: NextResponse.json(envelope, { status: failure.statusCode, headers }),
    envelope,
  };
}

export function withGatewayIngress<TParams extends Record<string, string> = Record<string, string>>(
  handler: (req: NextRequest, ctx: RequestContext, params: TParams) => Promise<Response>,
  options?: { skipReadinessCheck?: boolean },
) {
  return async (req: NextRequest, context: { params: Promise<TParams> }): Promise<Response> => {
    await connection();
    const requestId = randomUUID();
    try {
      const ctx = extractRequestContext(req, requestId);
      if (!options?.skipReadinessCheck) ensureGatewayReady();
      const resolvedParams = (await context?.params ?? {}) as TParams;
      const response = await handler(req, ctx, resolvedParams);
      response.headers.set(GATEWAY_HEADERS.REQUEST_ID, requestId);
      if (!response.headers.has('cache-control')) response.headers.set('cache-control', 'no-store');
      return response;
    } catch (error) {
      return formatErrorResponse(error, requestId).response;
    }
  };

export function loadGatewaySecretKey(db: GatewayDatabase): Buffer {
  const configured = process.env.GATEWAY_SECRET_KEY;
  if (configured) {
    if (!/^[A-Za-z0-9+/]{43}=$/.test(configured)) {
      throw new Error('GATEWAY_SECRET_KEY must be a base64-encoded 32-byte key.');
    }
    return Buffer.from(configured, 'base64');
  }

  if (process.env.NODE_ENV === 'production') {
    throw new Error('Set GATEWAY_SECRET_KEY before starting the gateway in production.');
  }
  if (db.path === ':memory:') return randomBytes(32);

  const keyPath = `${db.path}.key`;
  try {
    return readFileSync(keyPath);
  } catch (error) {
    if ((error as NodeJS.ErrnoException).code !== 'ENOENT') throw error;
  }

  const tempPath = `${keyPath}.${randomUUID()}.tmp`;
  try {
    writeFileSync(tempPath, randomBytes(32), { mode: 0o600, flag: 'wx', flush: true });
    try {
      linkSync(tempPath, keyPath);
    } catch (error) {
      if ((error as NodeJS.ErrnoException).code !== 'EEXIST') throw error;
    }
    return readFileSync(keyPath);
  } finally {
    rmSync(tempPath, { force: true });
  }
}

export class GatewaySecretStore implements ISecretStore {
  private readonly key: Buffer;

  constructor(private readonly db: GatewayDatabase, key: Buffer) {
    if (key.length !== 32) throw new Error('The gateway encryption key must have 32 bytes.');
    this.key = Buffer.from(key);
    this.db.write(() => {
      const marker = this.db.get<EncryptedSecret>('metadata', metadataScope, keyCheckId);
      if (marker) {
        try {
          if (this.decrypt(metadataScope, keyCheckId, marker) !== keyCheckId) throw new Error();
        } catch {
          throw new Error('The gateway encryption key does not match the saved data.');
        }
      } else {
        this.db.put('metadata', metadataScope, keyCheckId, this.encrypt(metadataScope, keyCheckId, keyCheckId));
      }
    });
  }

  private encrypt(scope: string, ref: string, value: string): EncryptedSecret {
    const nonce = randomBytes(12);
    const cipher = createCipheriv('aes-256-gcm', this.key, nonce);
    cipher.setAAD(Buffer.from(JSON.stringify([scope, ref])));
    const ciphertext = Buffer.concat([cipher.update(value, 'utf8'), cipher.final()]);
    return {
      nonce: nonce.toString('base64'),
      tag: cipher.getAuthTag().toString('base64'),
      ciphertext: ciphertext.toString('base64'),
    };
  }

  private decrypt(scope: string, ref: string, encrypted: EncryptedSecret): string {
    const decipher = createDecipheriv('aes-256-gcm', this.key, Buffer.from(encrypted.nonce, 'base64'));
    decipher.setAAD(Buffer.from(JSON.stringify([scope, ref])));
    decipher.setAuthTag(Buffer.from(encrypted.tag, 'base64'));
    return Buffer.concat([
      decipher.update(Buffer.from(encrypted.ciphertext, 'base64')),
      decipher.final(),
    ]).toString('utf8');
  }

  async getSecret(ref: string, ctx: RequestContext): Promise<string | null> {
    const encrypted = this.db.get<EncryptedSecret>('secrets', ctx.workspaceId, ref);
    return encrypted ? this.decrypt(ctx.workspaceId, ref, encrypted) : null;
  }

  async setSecret(ref: string, value: string, ctx: RequestContext): Promise<void> {
    this.db.write(() => this.writeSecret(ref, value, ctx));
  }

  async deleteSecret(ref: string, ctx: RequestContext): Promise<boolean> {
    return this.db.write(() => this.removeSecret(ref, ctx));
  }

  // Used by the connection store as part of its transaction.
  writeSecret(ref: string, value: string, ctx: RequestContext): void {
    this.db.put('secrets', ctx.workspaceId, ref, this.encrypt(ctx.workspaceId, ref, value));
  }

  removeSecret(ref: string, ctx: RequestContext): boolean {
    return this.db.delete('secrets', ctx.workspaceId, ref);
  }

        PRAGMA journal_mode = WAL;
        PRAGMA synchronous = FULL;
        CREATE TABLE IF NOT EXISTS gateway_documents (

          scope TEXT NOT NULL,
          id TEXT NOT NULL,
          value TEXT NOT NULL,
          timestamp INTEGER NOT NULL,
          PRIMARY KEY (collection, scope, id)

          ON gateway_documents (collection, scope, timestamp);
        CREATE TABLE IF NOT EXISTS gateway_events (
          scope TEXT NOT NULL,
          session_id TEXT NOT NULL,
          sequence INTEGER NOT NULL,
          value TEXT NOT NULL,
          PRIMARY KEY (scope, session_id, sequence)
        );
      `);
    } catch (error) {
      this.db.close();
      throw error;
    }
  }

    this.db.exec('BEGIN IMMEDIATE');
    try {
      const result = operation();
      this.db.exec('COMMIT');
      return result;
    } catch (error) {
      this.db.exec('ROLLBACK');
      throw error;
    }
  }

    this.db.prepare('SELECT 1 FROM gateway_documents LIMIT 1').get();
    return true;
  }

  close(): void {
    this.db.close();
  }

    const row = this.db.prepare(
      'SELECT value FROM gateway_documents WHERE collection = ? AND scope = ? AND id = ?'
    ).get(collection, scope, id);
    return row ? JSON.parse(String(row.value)) as T : null;
  }

    return this.db.prepare(`
      SELECT value FROM gateway_documents
      WHERE collection = ? AND scope = ? AND timestamp >= ? AND timestamp < ?
      ORDER BY timestamp DESC, id DESC
    `).all(collection, scope, since, until).map((row) => JSON.parse(String(row.value)) as T);

      VALUES (?, ?, ?, ?, ?)
      ON CONFLICT (collection, scope, id) DO UPDATE SET value = excluded.value, timestamp = excluded.timestamp

export class GatewayUsageService implements IUsageService {
  private readonly maxDailyTokens = readLimit('GATEWAY_MAX_DAILY_TOKENS') ?? 200_000;
  private readonly maxDailyBudgetUsd = readLimit('GATEWAY_MAX_DAILY_BUDGET_USD');

  constructor(private readonly db: GatewayDatabase) {}

  private recordsForWindow(ctx: RequestContext, now: Date): AttemptRecord[] {
    const { start, end } = utcDay(now);
    return this.db.list('attempts', ctx.workspaceId, start, end);
  }

  private commitments(records: AttemptRecord[]) {
    let tokens = 0;
    let costUsd = 0;
    for (const { reservation, accounting } of records) {
      if (accounting?.status === 'refunded') continue;
      tokens += accounting?.tokens?.total ?? reservation.estimatedTokens;
      costUsd += accounting?.costUsd ?? reservation.estimatedCostUsd ?? 0;
    }
    return { tokens, costUsd };
  }

  async reserveAttempt(input: ReservationInput, ctx: RequestContext): Promise<boolean> {
    if (input.workspaceId !== ctx.workspaceId || input.status !== 'reserved') return false;
    const estimatedTokens = Math.ceil(input.estimatedTokens ?? 8_192);
    if (!Number.isSafeInteger(estimatedTokens) || estimatedTokens < 1) return false;
    if (input.estimatedCostUsd !== undefined && (!Number.isFinite(input.estimatedCostUsd) || input.estimatedCostUsd < 0)) return false;

    return this.db.write(() => {
      const now = new Date();
      if (this.db.get<AttemptRecord>('attempts', ctx.workspaceId, input.attemptId)) return false;
      const { tokens, costUsd } = this.commitments(this.recordsForWindow(ctx, now));
      if (tokens + estimatedTokens > this.maxDailyTokens) return false;
      if (this.maxDailyBudgetUsd !== undefined &&
        (input.estimatedCostUsd === undefined || costUsd + input.estimatedCostUsd > this.maxDailyBudgetUsd)) return false;

      const record: AttemptRecord = {
        reservation: { ...input, estimatedTokens, createdAt: now.toISOString() },
      };
      this.db.put('attempts', ctx.workspaceId, input.attemptId, record, now.getTime());
      return true;
    });
  }

  async settleAttempt(input: AttemptAccounting, ctx: RequestContext): Promise<void> {
    this.db.write(() => {
      const record = this.db.get<AttemptRecord>('attempts', ctx.workspaceId, input.attemptId);
      if (!record || input.workspaceId !== ctx.workspaceId ||
        record.reservation.reservationId !== input.reservationId ||
        record.reservation.modelId !== input.modelId ||
        record.reservation.connectionId !== input.connectionId ||
        (record.reservation.runId !== undefined && record.reservation.runId !== input.runId)) {
        throw new GatewayError({ code: 'not_found', message: 'Attempt reservation not found', requestId: ctx.requestId });
      }
      if (record.accounting) return;
      if (input.status !== 'settled' && input.status !== 'refunded') {
        throw new GatewayError({ code: 'invalid_request', message: 'Attempt accounting needs a terminal status', requestId: ctx.requestId });
      }

      const tokens = input.tokens && Object.values(input.tokens).every((value) => Number.isSafeInteger(value) && value >= 0)
        ? input.tokens : undefined;
      const costUsd = typeof input.costUsd === 'number' && Number.isFinite(input.costUsd) && input.costUsd >= 0
        ? input.costUsd : undefined;
      record.accounting = {
        ...input, tokens, costUsd, settledAt: new Date().toISOString(),
      };
      this.db.put('attempts', ctx.workspaceId, input.attemptId, record, Date.parse(record.reservation.createdAt));
    });
  }

  private reconcileTerminalRuns(records: AttemptRecord[], ctx: RequestContext): void {
    for (const record of records) {
      if (record.accounting || !record.reservation.runId) continue;
      const stored = this.db.get<RunRecord>('runs', ctx.workspaceId, record.reservation.runId);
      // A completed handler can still deliver final metering briefly after publishing run state.
      if (!stored?.run.completedAt || isActive(stored.run.status) ||
        Date.now() - Date.parse(stored.run.completedAt) < 30_000) continue;

      const { attemptId, reservationId, workspaceId, runId, connectionId, modelId, createdAt } = record.reservation;
      record.accounting = {
        attemptId, reservationId, workspaceId, runId, connectionId, modelId,
        status: 'settled',
        durationMs: Math.max(0, Date.parse(stored.run.completedAt) - Date.parse(createdAt)),
        settledAt: new Date().toISOString(),
      };
      this.db.put('attempts', ctx.workspaceId, attemptId, record, Date.parse(createdAt));
    }
  }

  async getUsage(ctx: RequestContext): Promise<GatewayUsageSummary> {
    return this.db.write(() => {
      const now = new Date();
      const records = this.recordsForWindow(ctx, now);
      this.reconcileTerminalRuns(records, ctx);

      const { start, end } = utcDay(now);
      const { tokens, costUsd } = this.commitments(records);
      const totals = { promptTokens: 0, completionTokens: 0, totalTokens: 0, totalRequests: 0, estimatedCostUsd: 0 as number | undefined };
      const models = new Map<string, ModelUsageSummary>();
      let containsUnknownUsage = false;

      for (const { reservation, accounting } of records) {
        if (!accounting || accounting.status === 'refunded') continue;
        const modelId = reservation.modelId;
        const model = models.get(modelId) ?? { promptTokens: 0, completionTokens: 0, totalTokens: 0, requests: 0, estimatedCostUsd: 0 };
        const prompt = accounting.tokens?.prompt ?? 0;
        const completion = accounting.tokens?.completion ?? 0;
        const measured = accounting.tokens?.total ?? 0;
        const cost = accounting.costUsd ?? reservation.estimatedCostUsd;
        if (!accounting.tokens || accounting.costUsd === undefined) containsUnknownUsage = true;

        totals.promptTokens += prompt;
        totals.completionTokens += completion;
        totals.totalTokens += measured;
        totals.totalRequests++;
        totals.estimatedCostUsd = totals.estimatedCostUsd !== undefined && cost !== undefined ? totals.estimatedCostUsd + cost : undefined;
        model.promptTokens += prompt;
        model.completionTokens += completion;
        model.totalTokens += measured;
        model.requests++;
        model.estimatedCostUsd = model.estimatedCostUsd !== undefined && cost !== undefined ? model.estimatedCostUsd + cost : undefined;
        models.set(modelId, model);
      }
      if (records.some((record) => !record.accounting)) containsUnknownUsage = true;
      return {
        workspaceId: ctx.workspaceId,
        window: { start: new Date(start).toISOString(), end: new Date(end).toISOString() },
        freshness: containsUnknownUsage ? 'estimated' : 'fresh',
        totals,
        limits: {
          maxDailyTokens: this.maxDailyTokens,
          maxDailyBudgetUsd: this.maxDailyBudgetUsd,
          currentTokensUsed: tokens,
          currentBudgetUsdUsed: costUsd,
          isExceeded: tokens >= this.maxDailyTokens ||
            (this.maxDailyBudgetUsd !== undefined && costUsd >= this.maxDailyBudgetUsd),
        },
        byModel: Object.fromEntries(models),
        lastUpdatedAt: now.toISOString(),
      };
    });
  }
}
